\section{SQL Execution Order}
\key{Order:} FROM/JOIN → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT
\begin{itemize}
  \item WHERE: before grouping (no aggregates allowed here!)
  \item HAVING: after grouping (aggregates OK)
  \item Can't use SELECT alias in WHERE (alias not yet defined)
  \item Every non-agg col in SELECT must be in GROUP BY
  \item GROUP BY cols NOT in SELECT is valid
\end{itemize}

\section{SQL Patterns}
\key{Date overlap:} \sql{start\_date <= rangeEnd AND end\_date >= rangeStart}

\key{Zero-count outer join:} Use LEFT JOIN to include rows with no matches. Put filter on joined table INSIDE a subquery \emph{before} the join (not in outer WHERE). \qref{Sp26 MT1 Q1.1}
\begin{ansbox}
\sql{FROM teams t LEFT JOIN (SELECT * FROM games g WHERE g.season=1965 OR g.season=1966) ON t.id=g.homeTeamID}\\
\warn{NOT:} \sql{FROM teams LEFT JOIN games ON ... WHERE season=1965} (kills NULL rows)
\end{ansbox}

\key{"All X" pattern:}
\sql{GROUP BY s HAVING COUNT(DISTINCT x) = (SELECT COUNT(*) FROM X WHERE ...)}\\
\warn{Must use DISTINCT to avoid double-counting!} \qref{Fa24 MT1 Q1b}

\key{UNION removes dups; UNION ALL keeps them.}

\key{RIGHT JOIN + INNER JOIN = INNER JOIN} (NULLs from RIGHT eliminated by inner). \qref{Sp25 MT1 Q1c}

\key{FULL OUTER + WHERE NOT NULL rows = LEFT JOIN} effect.

\section{SQL Validity Traps}
(1) Non-agg SELECT col not in GROUP BY → \ans{invalid}\\
(2) Alias used in WHERE → \ans{invalid}\\
(3) Agg func in WHERE → \ans{invalid}\\
(4) HAVING without GROUP BY → \ans{invalid}\\
(5) GROUP BY without agg → \ans{VALID}\\
(6) \sql{COUNT(DISTINCT col)} skips NULLs → \ans{True}

\section{RA Operator Symbols}
$\proj$ = project (SELECT cols)\quad $\sel$ = select (WHERE filter)\\
$\agg_{g,\text{AGG}}$ = GROUP BY + aggregate\quad $\jn$ = join\quad $\ren$ = rename\\
$\times$ = cross product\quad $-$ = difference\quad $\cup$ = union

\key{Compilation order (inside-out, bottom-up):}
\begin{enumerate}
  \item Base tables (FROM)
  \item Cross product / joins (JOIN ... ON)
  \item Selections (WHERE) → $\sel$
  \item Group-by + agg (GROUP BY / AGG) → $\agg$
  \item Projection (SELECT cols) → $\proj$
  \item Rename aliases → $\ren$
\end{enumerate}

\section{SQL $\to$ RA Translation}
\key{Simple WHERE:}\\
\sql{SELECT title FROM Movies WHERE year>2000}\\
$\Rightarrow \proj_{\text{title}}\bigl(\sel_{\text{year}>2000}(\text{Movies})\bigr)$

\key{JOIN + GROUP BY + AVG:}\\
\sql{SELECT m.title, AVG(score) FROM Movies m JOIN Reviews r ON m.id=r.mid GROUP BY m.id, m.title}\\
$\Rightarrow \proj_{\text{title},\text{AVG}}\bigl(\agg_{m.id,m.title,\,\text{AVG(score)}}(Movies \jn_{m.id=r.mid} Reviews)\bigr)$
\begin{warnbox}
$\agg$ gets BOTH group-by cols AND the aggregate expression.\\
Outer $\proj$ keeps only the display columns.
\end{warnbox}

\key{LLM operator:} Treated as part of selection or projection condition.\\
\sql{WHERE LLM(...)="Yes"} compiles to $\sel_{LLM(...)=Yes}$\\
\sql{SELECT AVG(LLM(...))} compiles into the $\agg$ aggregate column.

\begin{ansbox}
Sp25 Q1.2 answer: Ex2 compiles to \textbf{B}:\\
$\proj_{m.title,AVG(LLM)}\bigl(\agg_{m.id,m.title,AVG(LLM)}(Movies \jn_{m.id=r.id} Reviews)\bigr)$
\end{ansbox}

\section{LLM Optimization Rules}
\key{Selection pushdown} = apply cheap filter first, THEN LLM. Fewer LLM calls. \qref{Q1.5→D}

\key{LLM pull-up} = move LLM selection later so cheap predicates run first. (Same result — reduces LLM calls since fewer rows reach LLM.) \qref{Q1.5→D}

\key{Pull-up through join: CANNOT if} LLM output is part of join condition. \qref{Q1.6→A}

\key{Cross join + pull-up costs MORE:} cross join inflates row count → more LLM calls. \qref{Q1.7→B}

\key{Pre-sort for caching:} Sort on the column INSIDE the LLM prompt string (e.g., \sql{movie\_info}) to get adjacent calls sharing a prefix. \qref{Q1.8→B}

\key{Hash vs sort for caching:} Hashing doesn't guarantee order within buckets → prefix-sharing LLM calls may NOT be adjacent. \qref{Q1.10→A,B}

\key{General QO techniques that help:} Selection pushdown ✓, Projection pushdown ✓. Selection pull-up ✗, Projection pull-up ✗, Materialization ✗. \qref{Q1.4→A,C}

\section{Selinger / System R Optimizer}
\begin{itemize}
  \item Only considers \textbf{left-deep} join trees (no bushy, no right-deep)
  \item Never uses Cartesian products unless forced (no join predicate between tables)
  \item \textbf{Pass $k$:} build all $k$-table left-deep plans from $(k{-}1)$-table plans
  \item Keeps cheapest plan per (subset, \textbf{interesting order}) pair
  \item \textbf{Interesting order:} a sort order useful for a later join, ORDER BY, or GROUP BY. \warn{Aggregates (SUM, AVG) do NOT create interesting orders.}
\end{itemize}

\key{3-table left-deep orders} for $\{A,B,C\}$:\\
$(A\jn B)\jn C$, $(B\jn A)\jn C$, $(A\jn C)\jn B$,\\
$(C\jn A)\jn B$, $(B\jn C)\jn A$, $(C\jn B)\jn A$\\
\warn{Not considered:} $A\jn(B\jn C)$ (right-deep)

\begin{ansbox}
Sp25 Q1.3: 3-way join Users, Reviews, Movies with WHERE m.genre="action". Considered by Selinger: \textbf{A only:} (Movies⋈Reviews)⋈Users.\\
Rejected: B,D = right-deep. C = Movies⋈Users has NO join pred → cross product.
\end{ansbox}

\section{Selectivity Formulas}
$\sel_{x=v}$: $1/V(x)$ \quad $\sel_{x>v}$: $(max-v)/(max-min)$ \quad $\sel_{x\text{ BETWEEN}a,b}$: $(b-a)/(max-min)$

AND: multiply selectivities. OR: add them (assume independence).

Join result size: $|R|\cdot|S|\,/\,\max(V(R,x),\,V(S,y))$

Ex: 4 sports, 10 names → \sql{sport=bball AND name=James}: $1/4 \cdot 1/10 = 1/40$

\section{External Merge Sort}
\key{Notation:} $N$ = data pages, $B$ = buffer pages.

\key{Pass 0 (create runs):} Read $B$ pages, sort in memory, write back.\\
$\Rightarrow \lceil N/B \rceil$ sorted runs, each $B$ pages long.

\key{Each merge pass:} Merges $B-1$ runs at a time (1 page per run + 1 output page).\\
Runs after pass $k$: $\lceil \text{prev}/({B-1}) \rceil$

\key{Total passes:}
\begin{ansbox}
$\#\text{passes} = 1 + \lceil\log_{B-1}\lceil N/B\rceil\rceil$ \qquad\qquad \textbf{Total I/O} $= 2N \times \#\text{passes}$
\end{ansbox}

\begin{ansbox}
$B \times (B-1) \ge N \;\Rightarrow\; B \approx \sqrt{N}$\\
Ex: N=400: $B(B-1)\ge400 \Rightarrow B=\mathbf{21}$ \qref{MT2 Q1.3g}
\end{ansbox}

\begin{ansbox}
N=503, B=17: P0→$\lceil503/17\rceil=30$ runs. P1→$\lceil30/16\rceil=2$ runs. P2→1. \textbf{3 passes. I/O=2·503·3=3018} \qref{Sp25 Final Q3}

\key{SMJ optimization (pipelining):} In the final sorting pass of a relation that produces $\le B-1$ sorted runs, stream directly into the merge-join step instead of writing to disk first.
\begin{ansbox}
T=4pp, P=50pp, B=4 (SMJ without opt):\\
Sort T: 1 pass → 8 I/Os (T fits in 1 run already! $\lceil4/4\rceil=1$)\\
Sort P: P0=$\lceil50/4\rceil=13$ runs → P1=$\lceil13/3\rceil=5$ → P2=$\lceil5/3\rceil=2$ → P3=1. \textbf{4 passes = 400 I/Os}\\
Merge: 4+50=54 I/Os. \textbf{Total = 462} \qref{MT2 Q2.3a}\\
Savings from opt: P has 2 runs after P2 → pipeline P3 into merge, saving $2\times50=\mathbf{100}$ I/Os \qref{MT2 Q2.3b}
\end{ansbox}

\section{Sort Trivia (T/F)}
\begin{itemize}
  \item Halving $B$ doubles I/O? \ans{F} — passes might not change
  \item Doubling $B$ always decreases I/O? \ans{F} — passes can stay same
  \item Sort cost depends on which column you sort on? \ans{F} — cost = $f(N,B)$ only \qref{Sp25 Q1.9}
\end{itemize}

\section{External Hashing}
\key{Partitioning phase:} Use $B-1$ buckets per pass. Reads + writes ALL data each pass.

\key{Conquer phase:} 1 final pass — read each partition into memory and hash/aggregate.

\key{Recursion:} After 1 partition pass, each bucket has $\approx N/(B-1)$ pages. Recurse until each bucket $\le B$ pages.

\key{Total passes (uniform hash):}
\begin{ansbox}
$\#\text{partition passes} = \lceil\log_{B-1}\lceil N/B\rceil\rceil$ \quad + 1 conquer pass\\
Total I/O $= 2N \times \#\text{passes}$ (read+write each pass)
\end{ansbox}

\begin{ansbox}
N=686, B=7: $\lceil\log_6\lceil686/7\rceil\rceil = \lceil\log_6 98\rceil = \lceil2.56\rceil = 3$ partition passes + 1 conquer = \textbf{4 passes total}. I/O = $2\times686\times4=5488$. But staff computed 6098 — check exact calc. \qref{Sp25 Q3}
\end{ansbox}

\key{Non-uniform partitions (key insight):} Each partition is handled independently. Larger partitions need more passes.

\begin{ansbox}
N=200, B=5, P1=100pp, P2=50pp, P3=30pp, P4=20pp after Pass 1:\\
P1 (100pp): $100/(4)=25$pp per sub-bucket → $25/4\approx7$pp → $7/4\approx2$pp → conquer. 4 more passes.\\
P2 (50pp): $50/4=13$pp → $13/4\approx4$pp → conquer. 3 more passes.\\
P3 (30pp) and P4 (20pp): each needs \textbf{same \#passes} = T. \qref{MT2 Q1.2d}\\
\textbf{Pass 2 writes:} P1 sub-partitions: 100pp. P2: 52pp. P3: 32pp. P4: 20pp. Total written = \textbf{204pp}. \qref{MT2 Q1.2c}\\
\textbf{Total I/Os for whole process:} 1924 (range 1901–2000). \qref{MT2 Q1.2e}
\end{ansbox}

\section{Hash Trivia (T/F)}
\begin{itemize}
  \item End of hashing: same-key records fit in memory together? \ans{T} — guaranteed within one partition
  \item Can use different hash function per page? \ans{T} — independent per partition
  \item Hash I/O suffers when: \ans{A (many dups)} and \ans{C (non-uniform hash)}
  \item Hash vs sort I/O: \ans{E — neither always better} (either can be more/less expensive)
  \item Hash can cost LESS than sort when: some partitions conquered early (fewer I/Os on those). \qref{MT2 Q1.1f}
  \item Hash can cost MORE when: bad hash → extra passes needed. \qref{MT2 Q1.1f}
\end{itemize}

\section{GHJ Min Buffer Formula}
\key{No recursive partitioning when:} Each partition of smaller table fits in $B-2$ pages (probe buffer).
\begin{ansbox}
$\dfrac{[T]}{B-1} \le B-2 \;\Rightarrow\; [T] \le (B-1)(B-2)$\\
T=4pp: $(B-1)(B-2)\ge4 \Rightarrow B=\mathbf{4}$ minimum \qref{MT2 Q2.2a}
\end{ansbox}

\key{GHJ fails if:} Any partition from BOTH tables with same key value can't fit in $B-2$ pages.
\begin{ansbox}
T=4pp (4rec/pg), P=50pp (8rec/pg), B=5: Need $>B-2=3$pp from BOTH tables with same key.\\
T: $\lceil4\rceil=4$ pages → $3$pp threshold → need $>12$ rec = $\ge13$ tuples with same key in T.\\
P: need $>3$pp = $>24$ rec in P → $\ge25$ tuples. Total min to guarantee failure: $13+25=\mathbf{38}$ \qref{MT2 Q2.2c}
\end{ansbox}

\section{Join I/O Cost Formulas}
\key{Notation:} $[R]$ = pages of R, $|R|$ = tuples in R, $B$ = buffer pages.

{\codefont\begin{tabular}{@{}l|l@{}}
\textbf{Join} & \textbf{I/O Cost} \\
\hline
SNLJ & $[R] + |R|\cdot[S]$ (R outer, 1 tuple at a time)\\
PNLJ & $[R] + [R]\cdot[S]$ (page at a time)\\
BNLJ & $[R] + \lceil[R]/(B{-}2)\rceil\cdot[S]$ (R outer)\\
INLJ & $[R] + |R|\cdot(\text{index lookup cost})$\\
SMJ  & $\text{sort}(R) + \text{sort}(S) + [R] + [S]$\\
GHJ  & $3([R]+[S])$ baseline (no recursive partition)\\
\end{tabular}}

\key{BNLJ:} $B-2$ pages outer, 1 page inner buffer, 1 output. Put SMALLER table outer.

\key{INLJ lookup costs:} Alt-2 = $h+1$ (traverse height + 1 data page). Alt-1 = $h$ (data IS in leaf). Unclustered Alt-2: up to 1 I/O per matching tuple.

\key{GHJ full formula:} Read both sides + write both sides (partition) + read both sides (probe) = 3 passes. If recursive: $+2([R]+[S])$ per extra partition pass.

\key{BHJ (Broadcast):} Broadcast small table to all nodes. Network $=|small|\times(M-1)$.

\section{Join True/False Gotchas}
\key{BNLJ:} \qref{MT2 Q2.1a}
\begin{itemize}
  \item More B always decreases BNLJ cost? \ans{F} — plateaus when entire outer fits in buffer ($[R]\le B-2$)
  \item Smaller outer never worse? \ans{F} — counterex: B=4, R=3pp, S=4pp: R-outer=11, S-outer=10
  \item Requires equality condition? \ans{F} — works on any predicate
  \item $[R]+2\le B \Rightarrow$ scan S exactly once? \ans{T}
  \item $B=3$: BNLJ = PNLJ? \ans{T} (block size = 1)
  \item $B=2$: BNLJ = PNLJ? \ans{F} (BNLJ needs $\ge3$ pages)
\end{itemize}

\key{INLJ:} \qref{MT2 Q2.1b}
\begin{itemize}
  \item Always cheaper than BNLJ/PNLJ/SNLJ? \ans{F} — None of the above always true
  \item Alt-1 vs clustered Alt-2: same I/O? \ans{F} (Alt-1 data in leaf, Alt-2 needs extra page)
  \item Clustered Alt-2 vs clustered Alt-3: same? \ans{F}
\end{itemize}

\section{GHJ Full Example}
\begin{ansbox}
T=4pp (4r/pg), P=50pp (8r/pg), B=5, join on staff\_id:\\
Partitioning: $B-1=4$ buckets.\\
T: 4R + $4$W (1pp each bucket) = 8\\
P: 50R + $\lceil50/4\rceil\cdot4=52$W = 102\\
Probe: read T partitions (4) + P partitions (52) = 56\\
\textbf{Total = 8 + 102 + 56 = 166 I/Os} \qref{MT2 Q2.2b}
\end{ansbox}

\section{Join Algorithm Cheat Code}
\begin{enumerate}
  \item Both sorted on join key → SMJ (no sort cost)
  \item Index on inner relation key → INLJ
  \item Outer fits in memory ($[R]\le B-2$) → BNLJ (1 pass over inner)
  \item Equality join, no interesting order → GHJ
  \item Default → BNLJ with smaller outer
\end{enumerate}

\section{B+ Tree Structure}
\key{Order $d$:}
\begin{itemize}
  \item Internal (non-root): $d$ to $2d$ keys; $d+1$ to $2d+1$ pointers
  \item Leaf: $d$ to $2d$ entries
  \item Root: 1 to $2d$ keys
\end{itemize}

\key{Height $h$:} search $= O(h)$. Lookup I/O $= h$ to reach leaf.

\key{Alternatives:}
\begin{itemize}
  \item \textbf{Alt-1:} actual record IN leaf. Always clustered. Only 1 Alt-1 per table.
  \item \textbf{Alt-2:} (key, RID) in leaf. +1 I/O to fetch data page. Project 2 uses Alt-2.
  \item \textbf{Alt-3:} (key, list of RIDs) in leaf. Good for high-cardinality duplicates.
\end{itemize}

\section{Buffer Policies}
\key{LRU:} Evict least recently used. Good for popular pages (hot set). \warn{Bad: sequential flood (scans larger than buffer).}

\key{MRU:} Evict most recently used. Good for sequential/repeated full-table scans.

\key{Clock:} Circular ring + ref bit per page. Access → set bit=1. Sweep: if bit=1→set 0, skip; if bit=0→evict.

\key{Random access:} LRU = MRU = any policy (no pattern to exploit). \qref{MT1 Q4; Fa24 MT1 Q3b-ii}

\key{Popular subset} (e.g., 40 hot pages with 50-page buffer): LRU wins — keeps hot pages. \qref{Fa24 MT1 Q3b-iii}

\key{No-reread query (single-pass):} All policies equal — nothing re-read anyway. \qref{Fa24 MT1 Q4g}

\section{LRU-K (MT2 Q4)}
\key{Idea:} Evict page whose K-th most recent access was furthest in the past (not just the 1st most recent).

\key{Why LRU-K for B+ trees?} Keeps frequently-accessed internal nodes (root!) in buffer longer than LRU. \textbf{Better} for leaf scan workloads. \qref{MT2 Q4.6}

\key{LRU-K always better than LRU?} \ans{False} — not for all workloads. \qref{MT2 Q4.5}

\key{Implementation change from LRU (linked list):}
\begin{ansbox}
Need to store K most recent access times per page. Evict page whose K-th most recent access is furthest in past. \textbf{Answer B.} \qref{MT2 Q4.4}
\end{ansbox}

\key{Clock approximation for LRU-K:} \qref{MT2 Q4.7}
\begin{itemize}
  \item Page enters buffer: \ans{initialize counter = 1}
  \item Page gets a hit: \ans{increment counter (cap at K)}
  \item Eviction check: evict if counter \ans{= 0}
  \item Clock hand passes page but no evict: \ans{decrement counter by 1}

\section{Deadlock Detection and Avoidance}

Mnemonic: old-waits-young-dies.

\key{Wound-Wait}:\\
If $T_i$ wants lock $T_j$ holds: $T_i$ higher priority → \textbf{wound} $T_j$ (T$_j$ aborts); $T_i$ lower priority → \textbf{wait}.\\
Mnemonic: old-wounds-young-waits.

\section{Buffer Policies $\to$ Log Needs}
\begin{itemize}
  \item \textbf{Steal:} uncommitted dirty pages CAN go to disk → need \textbf{UNDO} log
  \item \textbf{No-Steal:} can't write uncommitted → no UNDO needed (but big memory needed)
  \item \textbf{Force:} flush all dirty pages at commit → no REDO needed (but slow commits)
  \item \textbf{No-Force:} don't flush at commit → need \textbf{REDO} log
  \item \textbf{ARIES = Steal + No-Force} → needs \textbf{both UNDO and REDO}
\end{itemize}

\key{Undo logs OLD value; redo logs NEW value.} \qref{Fa24 Final Q1a→A}

\section{ARIES Data Structures}
\key{Log record fields:} LSN, prevLSN, type (UPDATE/CLR/COMMIT/ABORT/END), xactID, pageID, before-image, after-image.

\key{Dirty Page Table (DPT):} pageID $\to$ \textbf{recLSN} = LSN of FIRST log record that dirtied this page since last flush.

\key{Transaction Table (Xact Table):} xactID $\to$ (status: running/committing/aborting, \textbf{lastLSN})

\key{Key LSN fields:}
\begin{itemize}
  \item \textbf{pageLSN:} LSN of most recent log record affecting this page. In-memory $\ge$ on-disk.
  \item \textbf{recLSN:} first LSN to dirty page since last flush (only in DPT, set when page first dirtied).
  \item \textbf{lastLSN:} last log for a transaction (in Xact Table, updated on every log write).
  \item \textbf{undoNextLSN:} in a CLR — equals \textbf{prevLSN of the record being undone} (skip over it in undo). \qref{Fa24 Q1d→A}
\end{itemize}

\begin{warnbox}
recLSN = FIRST to dirty (stays until flushed). pageLSN = MOST RECENT update. They're different!\\
lastLSN = last log of a TRANSACTION. pageLSN = last log of a PAGE.
\end{warnbox}

\section{CLR (Compensation Log Record)}
\begin{itemize}
  \item Written DURING undo (normal abort OR recovery undo)
  \item \warn{NOT written during commit}
  \item Redo-only record (not undone again — prevents infinite undo loops)
  \item CLR is flushed BEFORE the actual undo operation applies to data
\end{itemize}

\begin{ansbox}
Fa24 Q1c: CLR written when: \ans{A (abort during normal op)} and \ans{B (abort during recovery)}\\
NOT C (flushing at commit) or D (checkpointing)
\end{ansbox}

\section{Checkpoint}
\key{Fuzzy checkpoint:} Don't pause new transactions.\\
Write BEGIN\_CKPT → snapshot DPT + Xact Table → write END\_CKPT.\\
\warn{Always write END\_CKPT, even if both tables are empty.} \qref{Pj5 fact}

\key{Non-fuzzy:} Pause all new TX\_START during snapshot. Clean but slow.

\key{Multiple END\_CKPT:} Possible if tables too large for one log record.

\section{ARIES Recovery — 3 Phases}

\begin{greenbox}
\textbf{Phase 1 — Analysis:} Scan forward from last BEGIN\_CKPT to end of log.\\
• On UPDATE: if page not in DPT → add (recLSN = this LSN). Update xact's lastLSN.\\
• On COMMIT: update xact status to "committing".\\
• On END: remove xact from Xact Table.\\
After analysis: xacts still "committing" → write END, remove. All remaining xacts → status = "aborting" (losers to undo).
\end{greenbox}

\begin{greenbox}
\textbf{Phase 2 — Redo:} Start from \textbf{min(recLSN across all DPT entries)}.\\
Redo every UPDATE and CLR, \textbf{UNLESS any of:}\\
(a) Affected page is NOT in DPT.\\
(b) Log LSN $<$ page's recLSN in DPT (page was flushed and re-dirtied later).\\
(c) On-disk pageLSN $\ge$ this log's LSN (change already persisted).
\end{greenbox}

\begin{greenbox}
\textbf{Phase 3 — Undo:} Undo all loser (aborting) transactions.\\
• Maintain a \textbf{max-heap} (priority queue by LSN descending) of LSNs to undo.\\
• Initialize: all lastLSNs of loser xacts.\\
• Pop max LSN: if UPDATE → write CLR + apply undo, push prevLSN of that record.\\
• If CLR → push its undoNextLSN (skip over already-undone record).\\
• If prevLSN/undoNextLSN is null → xact fully undone, write END.
\end{greenbox}

\section{2-Phase Commit (2PC) Protocol}
\key{Roles:} Coordinator $C$ + Participants $P_1\ldots P_n$

\begin{greenbox}
\textbf{Steps:}\\
1. $C$ sends PREPARE to all $P_i$\\
2. Each $P_i$: log+flush "prepare" record, reply VOTE-YES or VOTE-NO\\
3a. All YES: $C$ logs+flushes COMMIT → sends COMMIT to all\\
   Each $P_i$: log+flush COMMIT, send ACK\\
   $C$ logs END after all ACKs received\\
3b. Any NO or timeout: $C$ logs ABORT → sends ABORT to all
\end{greenbox}

\key{Crash Scenarios:} \qref{Fa24 Q1e}
\begin{itemize}
  \item $P_i$ crashes after voting YES: sees "prepare" in log → send STATUS\_INQUIRY to $C$, \textbf{wait} (BLOCKING)
  \item $C$ crashes BEFORE logging COMMIT: no commit in log → \textbf{ABORT} on recovery
  \item $C$ crashes AFTER logging COMMIT: commit in log → \textbf{resend COMMIT} on recovery
  \item $C$ log EMPTY (no commit logged): \textbf{locally ABORT immediately} \qref{Fa24 Q1e→D}
\end{itemize}

\key{Presumed Abort:} If $C$ has no record of a transaction upon inquiry → assume ABORT.
\begin{ansbox}
Benefits of presumed abort vs no-presumed-abort: \qref{Fa24 Q1f→B}\\
\ans{B: Reduces network data by not requiring ACKs for abort messages.}\\
NOT A (more fault tolerance — false), NOT D (reduces PREPARE disk writes — false)
\end{ansbox}

\key{Timing / earliest flush:} After $C$ sends PREPARE, earliest $P_i$ flush = shortest latency + 10ms to log.

\key{2PC is BLOCKING:} If coordinator fails after PREPARE but before COMMIT, all participants block waiting.

\section{Parallel Query Processing (PQP)}
\key{Architectures:}
\begin{itemize}
  \item \textbf{Shared Memory:} all CPUs share one RAM. Easy, but doesn't scale beyond small clusters.
  \item \textbf{Shared Disk:} separate RAM per node, shared persistent storage. Medium scale.
  \item \textbf{Shared Nothing:} own CPU + RAM + disk per node. Best scale; requires network communication.
\end{itemize}

\key{Data Partitioning:}
\begin{itemize}
  \item \textbf{Range:} split on key ranges. Good for range queries. Risk: data skew.
  \item \textbf{Hash:} hash(key) mod $N$. Even distribution. NO range help.
  \item \textbf{Round-robin:} perfectly even. Bad for ANY predicate pushdown.
\end{itemize}

\key{Hash partition network cost} ($N$ pages, $M$ machines):
\begin{ansbox}
Each page sent to exactly one other machine. Each machine keeps $1/M$, sends $(M-1)/M$.\\
\textbf{Total network cost} $= N \times (M-1)/M$ page-transfers\\
Ex: N=888pp, M=4, 1KiB/pp: $888 \times 3/4 = 666$ KiB total network. \qref{Sp25 Q7}
\end{ansbox}

\key{Broadcast Hash Join:} Small table broadcast to ALL nodes. Network $= |\text{small}| \times (M-1)$. Use when one table is tiny (e.g., 10 rooms vs. millions of measurements).

\section{MapReduce}
\key{Pipeline:} Map → (write to disk) → Shuffle+Sort by key → Reduce

\key{True/False:} \qref{Fa24 Final Q1k}
\begin{itemize}
  \item Intermediate results always written to disk: \ans{A = True} (fault tolerance requirement)
  \item Intermediate may stream map→reduce: \ans{B = False}
  \item Stragglers mitigated by speculative/backup execution: \ans{C = True}
  \item More I/O than rookieDB for SQL: \ans{D = True}
\end{itemize}

\key{Reducer fails (no straggler opt):} \qref{Fa24 Q1l→B}\\
\ans{B: Only that reducer's data is reprocessed by another node.}\\
NOT A (whole job fails), NOT C (restart from beginning)

\key{Key property:} MapReduce ALWAYS sorts by key between map and reduce (shuffle phase).

\section{Spark}
\key{True/False:} \qref{Fa24 Final Q1m}
\begin{itemize}
  \item Lazy execution → whole DAG optimized before executing: \ans{A = True}
  \item Higher I/O than MapReduce: \ans{B = False} (Spark has LOWER I/O — fewer disk writes)
  \item Lineage (RDD) recomputes lost partitions from source: \ans{C = True}
  \item Server fail → always restart entire program: \ans{D = False} (lineage avoids full restart)
\end{itemize}

\key{When Spark MUST restart:} If lost partition has no lineage (no parent RDD) AND no materialized checkpoint. Otherwise re-derive from source.

\section{NoSQL / BASE Properties}
\key{OLAP vs OLTP:} \qref{Fa24 Q1h}
\begin{itemize}
  \item OLAP (analytics): Inventory forecasting, website traffic analysis → \ans{A, C}
  \item OLTP (transactions): Banking processing, e-commerce cart management → \ans{B, D}
\end{itemize}

\key{BASE (vs ACID):} \qref{Fa24 Q1i→A,B,C}
\begin{itemize}
  \item \textbf{B}asic Availability: system always responds (may return stale data, but no errors unless invalid input) → \ans{A = True}
  \item \textbf{S}oft State: state may change over time even without input (due to eventual consistency propagation) → \ans{B = True}
  \item \textbf{E}ventual Consistency: reads may see outdated data until updates propagate → \ans{C = True}
  \item D is FALSE — state change is NOT from pseudorandom mechanisms
\end{itemize}

\key{Semi-structured vs Relational:} \qref{Fa24 Q1j→A}
\begin{itemize}
  \item No predefined schema required → \ans{A = True}
  \item Semi-structured is binary? → \ans{B = False} (both usually text-based, e.g. JSON/XML)
  \item Can map relational→JSON? → Yes → \ans{C = False}
  \item Can map JSON→relational? → Yes → \ans{D = False}
\end{itemize}
